from pathlib import Path
import numpy as np
from PIL import Image
O=Path(__file__).resolve().parent;T=O/'Textures';T.mkdir(exist_ok=True)
n=1024;rng=np.random.default_rng(9021)
def noise(k):
 a=rng.standard_normal((n,n));freq=np.fft.fftfreq(n)*n;fall=np.exp(-(freq[:,None]**2+freq[None,:]**2)/(k*k/8));a=np.fft.ifft2(np.fft.fft2(a)*fall).real;return a/(a.std()*6)
f=noise(6)*.45+noise(20)*.28+noise(70)*.17+noise(230)*.07+noise(1024)*.03
for name,col,rough,amp in [('Fieldstone2m',(.36,.34,.29),.83,.0018),('LimeMortar2m',(.43,.415,.36),.94,.0007),('Soot2m',(.045,.039,.031),.97,.00035)]:
 h=(f+noise(450)*.055);bc=np.clip(np.array(col)[None,None,:]+f[:,:,None]*(.22 if name!='Soot2m' else .035),0,1)
 # Fine pits and mineral flecks at millimetre scale; no baked illumination.
 pits=rng.random((n,n))<.013;bc-=pits[:,:,None]*(.045 if name!='Soot2m'else .006)
 dx=(np.roll(h,-1,axis=1)-np.roll(h,1,axis=1))*amp*n/4;dy=(np.roll(h,-1,axis=0)-np.roll(h,1,axis=0))*amp*n/4
 normal=np.stack([-dx,dy,np.ones_like(h)],2);normal/=np.linalg.norm(normal,axis=2)[:,:,None]
 orm=np.stack([np.ones_like(h),np.clip(rough+f*.12,.6,1),np.zeros_like(h)],2)
 for suffix,a in [('BC',bc),('N_GL',normal*.5+.5),('ORM',orm)]:Image.fromarray((np.clip(a,0,1)*255+.5).astype('uint8')).save(T/f'T_Shared_{name}_{suffix}.png')
print('9 texture maps written')
